from sentence_transformers import SentenceTransformer
from sklearn.metrics.pairwise import cosine_similarity

# 1. 加载本地 bge 模型
model_path = "./BAAI/bge-small-zh-v1.5"
model = SentenceTransformer(model_path)

# 2. 定义待检索文本和候选数据库文本
query = "我今天很开心"
corpus = [
    "我喜欢机器学习",
    "我喜欢深度学习",
    "我今天心情很不错"
]

# 3. 将文本转换为向量（Embedding）
query_embedding = model.encode(query).reshape(1, -1)
corpus_embeddings = model.encode(corpus)

# 4. 计算 query 与每条语料的余弦相似度
similarity_scores = cosine_similarity(query_embedding, corpus_embeddings)[0]

# 5. 按相似度从高到低排序并输出结果
ranked_results = sorted(
    zip(corpus, similarity_scores),
    key=lambda x: x[1],
    reverse=True
)

print("===== 文本检索结果 =====")
print(f"待检索文本：{query}\n")
for rank, (text, score) in enumerate(ranked_results, 1):
    print(f"第{rank}名 | 相似度 {score:.4f} | {text}")
