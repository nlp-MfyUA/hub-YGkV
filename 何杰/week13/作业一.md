import os
import sqlite3
from typing import TypedDict, Optional
from typing_extensions import Annotated
from langgraph.graph import StateGraph, START, END
from langgraph.graph.message import add_messages
from langchain_openai import ChatOpenAI
from langchain_community.utilities import SQLDatabase

os.environ["OPENAI_API_KEY"] = "sk-78cc4e9ac8f44efdb207b7232e1ae6d8"

MODEL_NAME = "qwen-turbo"  # 可换成 deepseek-flash

# 加载 chinook.db，放到同目录
db = SQLDatabase.from_uri("sqlite:///chinook.db")
print("✅ 数据库加载成功，表列表：", db.get_usable_table_names())

llm = ChatOpenAI(model=MODEL_NAME, temperature=0)

# ========== 2. 定义Graph State（状态，全程流转） ==========
class SqlAgentState(TypedDict):
    messages: Annotated[list, add_messages]
    user_query: str                # 用户自然语言问题
    db_schema: Optional[str]       # 数据库全局解析信息
    candidate_tables: Optional[str]# 筛选出来的相关表
    sql_raw: Optional[str]         # 初次生成SQL
    sql_valid: Optional[bool]      # SQL校验是否通过
    sql_fixed: Optional[str]       # 改写后的SQL
    exec_result: Optional[str]     # SQL执行结果
    final_answer: Optional[str]    # 最终自然语言回答
    error_msg: Optional[str]       # 错误信息

# ========== 3. 定义7个节点函数 ==========
def node_db_parse(state: SqlAgentState):
    """节点1：数据库解析，获取全库表结构"""
    schema_info = db.get_table_info()
    return {
        "db_schema": schema_info,
        "messages": [("system", f"【数据库解析完成】schema:\n{schema_info}")]
    }

def node_table_select(state: SqlAgentState):
    """节点2：根据用户问题，挑选相关表"""
    prompt = f"""
你是数据库专家。
用户问题：{state['user_query']}
数据库所有表结构：
{state['db_schema']}
只输出和问题相关的表名，用逗号分隔，不要多余解释。
"""
    resp = llm.invoke(prompt)
    tables = resp.content.strip()
    return {
        "candidate_tables": tables,
        "messages": [("system", f"【筛选表】{tables}")]
    }

def node_sql_generate(state: SqlAgentState):
    """节点3：生成SQL，只允许SELECT"""
    table_schema = db.get_table_info(state["candidate_tables"].split(","))
    prompt = f"""
基于下面表结构，生成sqlite SELECT语句回答用户问题。
禁止任何修改语句：INSERT/UPDATE/DELETE/DROP/ALTER。
表结构：
{table_schema}
用户问题：{state['user_query']}
只输出SQL，不要解释，不要```标记。
"""
    resp = llm.invoke(prompt)
    sql = resp.content.strip()
    return {
        "sql_raw": sql,
        "sql_fixed": sql,
        "messages": [("system", f"【生成SQL】{sql}")]
    }

def node_sql_verify(state: SqlAgentState):
    """节点4：SQL校验，语法、字段、表是否合法"""
    sql = state["sql_fixed"]
    prompt = f"""
校验这条sqlite SQL：{sql}
已知可用表：{state['candidate_tables']}
检查：语法错误、表不存在、字段不存在、是否为只读SELECT。
输出JSON格式：{{"valid":true/false,"reason":"原因"}}
"""
    resp = llm.invoke(prompt)
    content = resp.content.strip()
    import json
    try:
        j = json.loads(content)
        valid = j["valid"]
        reason = j["reason"]
    except:
        valid = False
        reason = "LLM校验返回格式异常"
    return {
        "sql_valid": valid,
        "error_msg": reason,
        "messages": [("system", f"【校验结果】valid={valid}, reason={reason}")]
    }

def node_sql_fix(state: SqlAgentState):
    """节点5：SQL改写修复（校验失败进入）"""
    table_schema = db.get_table_info(state["candidate_tables"].split(","))
    prompt = f"""
原始SQL：{state['sql_fixed']}
错误原因：{state['error_msg']}
表结构：{table_schema}
用户问题：{state['user_query']}
修正SQL，只能SELECT，只输出SQL文本。
"""
    resp = llm.invoke(prompt)
    new_sql = resp.content.strip()
    return {
        "sql_fixed": new_sql,
        "messages": [("system", f"【重写SQL】{new_sql}")]
    }

def node_sql_execute(state: SqlAgentState):
    """节点6：执行SQL"""
    try:
        res = db.run(state["sql_fixed"])
        return {
            "exec_result": res,
            "messages": [("system", f"【执行成功】结果：{res}")]
        }
    except Exception as e:
        return {
            "exec_result": f"执行异常:{str(e)}",
            "sql_valid": False,
            "error_msg": str(e),
            "messages": [("system", f"【执行失败】{str(e)}")]
        }

def node_summary(state: SqlAgentState):
    """节点7：结果汇总，自然语言回答"""
    prompt = f"""
用户问题：{state['user_query']}
执行SQL：{state['sql_fixed']}
查询结果：{state['exec_result']}
用中文整理成通顺回答。
"""
    resp = llm.invoke(prompt)
    return {
        "final_answer": resp.content,
        "messages": [("ai", resp.content)]
    }

# ========== 4. 条件路由函数：校验失败就回到改写节点 ==========
def route_after_verify(state: SqlAgentState) -> str:
    if state["sql_valid"] is True:
        return "sql_execute"
    else:
        return "sql_fix"

# ========== 5. 构建LangGraph工作流 ==========
builder = StateGraph(SqlAgentState)
# 添加节点
builder.add_node("db_parse", node_db_parse)
builder.add_node("table_select", node_table_select)
builder.add_node("sql_generate", node_sql_generate)
builder.add_node("sql_verify", node_sql_verify)
builder.add_node("sql_fix", node_sql_fix)
builder.add_node("sql_execute", node_sql_execute)
builder.add_node("summary", node_summary)

# 流程连线
builder.add_edge(START, "db_parse")
builder.add_edge("db_parse", "table_select")
builder.add_edge("table_select", "sql_generate")
builder.add_edge("sql_generate", "sql_verify")
builder.add_conditional_edges("sql_verify", route_after_verify, {
    "sql_execute": "sql_execute",
    "sql_fix": "sql_fix"
})
builder.add_edge("sql_fix", "sql_verify") # 修复完重新校验
builder.add_edge("sql_execute", "summary")
builder.add_edge("summary", END)

# 编译图
graph = builder.compile()

# ========== 6. 调用测试 ==========
if __name__ == "__main__":
    user_input = "哪个国家的客户消费总额最高？"
    result = graph.invoke({
        "user_query": user_input
    })
    print("\n==== Final Answer ====")
    print(result["final_answer"])
    print("\n==== 执行SQL ====")
    print(result["sql_fixed"])
