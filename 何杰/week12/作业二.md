# ========== 2. 工具函数：数据库解析，获取表结构 ==========
def get_db_schema() -> str:
    """读取 sqlite 所有表+字段信息"""
    conn = sqlite3.connect(DB_PATH)
    cur = conn.cursor()
    cur.execute("SELECT name FROM sqlite_master WHERE type='table';")
    tables = [t[0] for t in cur.fetchall()]
    schema_text = ""
    for table in tables:
        cur.execute(f"PRAGMA table_info({table});")
        cols = cur.fetchall()
        schema_text += f"【表名】{table}\n"
        for c in cols:
            cid, name, dtype, notnull, dflt, pk = c
            schema_text += f"  - {name} ({dtype}) {'主键' if pk else ''}\n"
        schema_text += "\n"
    conn.close()
    return schema_text

# ========== 3. Pydantic 输出结构定义（LangChain JsonParser） ==========
class TableSelectOutput(BaseModel):
    related_tables: List[str] = Field(description="和用户问题相关的表名列表")
    reason: str = Field(description="选择这些表的理由")

class SqlGenerateOutput(BaseModel):
    sql: str = Field(description="生成的SQL查询语句，只允许SELECT")
    explain: str = Field(description="这条SQL的说明")

class SqlCheckOutput(BaseModel):
    pass_check: bool = Field(description="SQL是否通过校验")

    fixed_sql: str = Field(description="修正后的SQL")
    modify_reason: str = Field(description="修改说明")

# 4.5 SQL执行函数（安全执行）
def execute_sql(sql: str):
    """安全执行select，只允许查询"""
    if not sql.strip().lower().startswith("select"):
        return {"ok": False, "data": None, "err": "禁止非SELECT语句"}
    try:
        conn = sqlite3.connect(DB_PATH)
        cur = conn.cursor()
        cur.execute(sql)
        rows = cur.fetchall()
        cols = [desc[0] for desc in cur.description]
        res = [dict(zip(cols, r)) for r in rows]
        conn.close()
        return {"ok": True, "data": res, "err": None}
    except Exception as e:
        return {"ok": False, "data": None, "err": str(e)}

# ========== 5. 主流程，串联完整 workflow ==========
def sql_agent_run(user_query: str):
    print("===== 步骤1：读取数据库schema =====")
    db_schema = get_db_schema()
    print("===== 步骤2：选择相关表 =====")
    table_out = table_select_chain.invoke({
        "db_schema": db_schema,
        "user_query": user_query
    })
    print("选中表：", table_out["related_tables"])

    print("===== 步骤3：生成初始SQL =====")
    gen_out = sql_gen_chain.invoke({
        "db_schema": db_schema,
        "related_tables": table_out["related_tables"],
        "user_query": user_query
    })
    raw_sql = gen_out["sql"]
    print("初始SQL：", raw_sql)

    print("===== 步骤4：SQL校验 =====")
    check_out = sql_check_chain.invoke({"sql": raw_sql})
    print("校验结果：", check_out)

    if not check_out["pass_check"]:
        print("===== 步骤5：校验不通过，改写SQL =====")
        rewrite_out = sql_rewrite_chain.invoke({
            "db_schema": db_schema,
            "related_tables": table_out["related_tables"],
            "sql": raw_sql,
            "check_msg": check_out["message"]
        })
        final_sql = rewrite_out["fixed_sql"]
        print("改写后SQL：", final_sql)
    else:
        final_sql = raw_sql
        print("校验通过，无需改写")

    print("===== 步骤6：执行SQL =====")
    exec_result = execute_sql(final_sql)
    print("执行结果：", exec_result)

    print("===== 步骤7：结果汇总自然语言 =====")
    if exec_result["ok"]:
        ans = summary_chain.invoke({
            "user_query": user_query,
            "final_sql": final_sql,
            "query_result": json.dumps(exec_result["data"], ensure_ascii=False)
        })
    else:
        ans = f"SQL执行失败：{exec_result['err']}"

    return {
        "user_query": user_query,
        "related_tables": table_out,
        "origin_sql": raw_sql,
        "check": check_out,
        "final_sql": final_sql,
        "exec": exec_result,
        "answer": ans.content if hasattr(ans, "content") else ans
    }


if __name__ == "__main__":
