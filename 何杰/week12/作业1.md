RISK_PROMPT = """
你是消防安全检查员，识别图片内容，严格按照下面规则判断：
类别枚举只能是下面5个之一：["无风险", "低风险", "中等风险", "高风险", "非楼道"]
判定规则：

5. 非楼道：图片场景不是楼道（室外、卧室、厨房、马路等）。

输出要求：只返回标准JSON，禁止额外文字、解释。JSON字段：
{
    "scene_type": "",
    "risk_level": "",
    "risk_detail": "",
    "reason": ""
}
scene_type：描述场景，例如“居民楼道”、“室外马路”
risk_level：只能从上面5个里面选一个
risk_detail：列出识别到的隐患物品，无隐患填空数组
reason：一句话说明判定理由
"""

class DetectResult(BaseModel):
    scene_type: str
    risk_level: str
    risk_detail: list[str]
    reason: str


def image_to_base64(file_bytes: bytes) -> str:
    return base64.b64encode(file_bytes).decode("utf-8")


def call_vl_model(img_b64: str):
    if MODEL_CHOICE == "ds":
        headers = {"Authorization": f"Bearer {DS_API_KEY}", "Content-Type": "application/json"}
        payload = {
            "model": "deepseek-flash-4.1",
            "messages": [
                {
                    "role": "user",
                    "content": [
                        {"type": "text", "text": RISK_PROMPT},
                        {"type": "image_url", "image_url": {"url": f"data:image/jpeg;base64,{img_b64}"}}
                    ]
                }
            ],
            "temperature": 0.05
        }
    else:
        headers = {"Authorization": f"Bearer {QWEN_API_KEY}", "Content-Type": "application/json"}
        payload = {
            "model": "qwen-vl-max",
            "messages": [
                {
                    "role": "user",
                    "content": [
                        {"type": "text", "text": RISK_PROMPT},
                        {"type": "image_url", "image_url": {"url": f"data:image/jpeg;base64,{img_b64}"}}
                    ]
                }
            ],
            "temperature": 0.05
        }
    resp = requests.post(DS_URL if MODEL_CHOICE=="ds" else QWEN_URL, headers=headers, json=payload, timeout=60)
    resp.raise_for_status()
    content = resp.json()["choices"][0]["message"]["content"]
    return json.loads(content)


@app.post("/detect", response_model=DetectResult)
async def detect(file: UploadFile = File(...)):
    bytes_data = await file.read()
    b64_img = image_to_base64(bytes_data)
    result = call_vl_model(b64_img)
    return result

<script>
const video = document.getElementById('video');
const canvas = document.getElementById('canvas');
const ctx = canvas.getContext('2d');
const startBtn = document.getElementById('startBtn');
const shotBtn = document.getElementById('shotBtn');
const uploadBtn = document.getElementById('uploadBtn');
const previewBox = document.getElementById('preview');
const resultBox = document.getElementById('result');

let stream = null;
let photoBlob = null;

startBtn.onclick = async ()=>{
    try{
        stream = await navigator.mediaDevices.getUserMedia({
            video: { facingMode: "environment" }
        });
        video.srcObject = stream;
        shotBtn.disabled = false;
    }catch(err){
        alert("摄像头权限失败：本地file协议大多不支持，需要localhost/HTTPS\n"+err.message);
    }
}

shotBtn.onclick = ()=>{
    canvas.width = video.videoWidth;
    canvas.height = video.videoHeight;
    ctx.drawImage(video,0,0);
    canvas.toBlob(blob=>{
        photoBlob = blob;
        previewBox.innerHTML = `<img class="preview-img" src="${URL.createObjectURL(blob)}">`;
        uploadBtn.disabled = false;
    }, "image/jpeg", 0.82);
}

uploadBtn.onclick = async ()=>{
    if(!photoBlob) return;
    resultBox.innerHTML = "<div>识别中，请稍候...</div>";
    const fd = new FormData();
    fd.append("file", photoBlob, "capture.jpg");
    try{
        const res = await fetch("http://127.0.0.1:8000/detect", {
            method:"POST",
            body: fd
        });
        const data = await res.json();
        let cssClass = "risk-other";
        if(data.risk_level === "高风险") cssClass = "risk-high";
        else if(data.risk_level === "中等风险") cssClass = "risk-mid";
        else if(data.risk_level === "低风险") cssClass = "risk-low";
        else if(data.risk_level === "无风险") cssClass = "risk-safe";

        resultBox.innerHTML = `
        <div class="result-card ${cssClass}">
            <h3>风险等级：${data.risk_level}</h3>
            <p><b>场景：</b>${data.scene_type}</p>
            <p><b>识别隐患：</b>${data.risk_detail.join("、") || "无"}</p>
            <p><b>判定理由：</b>${data.reason}</p>
        </div>`
    }catch(e){
        resultBox.innerHTML = `<div style="color:red">请求出错：${e}</div>`
    }
}
</script>
