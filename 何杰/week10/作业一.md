# ===================== 定义输出结构（强制固定JSON） =====================
class TranslateResult(BaseModel):
    """作业要求输出的JSON结构"""
    query_input: str
    language: str
    result: str

# ===================== 专家翻译Agent =====================
# 中文翻译成英文agent
zh_to_en_agent = Agent(
    name="中文翻译英文专家",
    model="qwen-max",
    handoff_description="当检测到输入是中文文本时，使用该代理，把中文翻译为地道英文。",
    instructions="你是翻译专家，将用户输入中文翻译成通顺英文。输出严格按照TranslatResult结构返回json，不要多余文字。",
    output_type=TranslateResult
)

# 外文翻译成中文agent
other_to_zh_agent = Agent(
    name="外文翻译中文专家",
    model="qwen-max",
    handoff_description="当检测到输入不是中文（英文/日语/韩语等），使用该代理，把文本翻译成中文。",
    instructions="你是翻译专家，将非中文文本翻译成通顺中文。输出严格按照TranslateResult结构返回json，不要多余文字。",
    output_type=TranslateResult
)

# ===================== 路由Triage Agent （判断语种，分发任务） =====================
triage_translate_agent = Agent(
    name="语种识别路由Agent",
    model="qwen-max",
    instructions="""你的任务：识别用户输入文本是什么语种。
    如果识别结果是【中文】，就交给 中文翻译英文专家；
    如果是其他语言（英文、日语、法语等所有非中文），交给外文翻译中文专家。
    """,
    handoffs=[zh_to_en_agent, other_to_zh_agent],
)


async def main():
    print("==== 语种识别互译Agent（OpenAI Agents SDK）====\n")
    test_cases = [
        "今天天气很不错",
        "Hello, nice to meet you",
        "私はコーヒーが好きです"
    ]

    for text in test_cases:
        print("="*60)
        print(f"用户输入：{text}")
        res = await Runner.run(triage_translate_agent, text)
        output = res.final_output_as(TranslateResult)
        print(f"\n✅ 输出JSON对象：")
        print(output.model_dump_json(indent=4, ensure_ascii=False))

if __name__ == "__main__":
    asyncio.run(main())
    try:
        draw_graph(triage_translate_agent, filename="lang_trans_agent")
    except Exception as e:
        print("\n绘制流程图跳过：", e)
