import random
import math
from fastmcp import FastMCP
from typing import Union

mcp = FastMCP(name="Dice Roller")


@mcp.tool
def roll_dice(n_dice: int) -> list[int]:
    """Roll `n_dice` 6-sided dice and return the results."""
    return [random.randint(10, 60) for _ in range(n_dice)]

@mcp.tool
def get_news() -> Union[int, str]:
    """Get latest news."""
    return "最新消息 Qwen4 is coming!"

# ========= 新增3个MCP工具 =========
@mcp.tool
def calc_area_circle(radius: float) -> float:
    """
    计算圆的面积
    Args:
        radius: 圆的半径
    Returns:
        圆面积结果
    """
    return math.pi * radius * radius


@mcp.tool
def get_random_poem() -> str:
    """随机返回一句中国古诗"""
    poem_list = [
        "春风又绿江南岸，明月何时照我还",
        "举头望明月，低头思故乡",
        "欲穷千里目，更上一层楼",
        "海内存知己，天涯若比邻"
    ]
    return random.choice(poem_list)


@mcp.tool
def convert_celsius_fahrenheit(c: float) -> float:
    """
    摄氏度转换为华氏度
    Args:
        c: 摄氏度数值
    Returns:
        转换后的华氏温度
    """
    return c * 9 / 5 + 32

    mcp.run(transport="sse", port=8000)
